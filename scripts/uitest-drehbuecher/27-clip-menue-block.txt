# P8-L Vocal-Tuning, P7a-L Tempo-Map.
#
# Messgroesse ist durchweg der NAME des Clips samt Rechteck (aus find) bzw. die
# angezeigte BPM. Was gerendert wird, ist engine-belegt (Bounce 163 dB Abstand zum
# Live-Render, Tuning 445 Hz → 440 Hz, Merge Rest exakt null) — hier zaehlt, dass die
# Geste im Programm ankommt, den Clip TAUSCHT und genau EIN Undo zurueckfuehrt.
#
# Zwei Dateien noetig:
#   AIZANA_TESTKURZ  — kurzer Ausschnitt (Merge/Bounce/Tuning)
#   AIZANA_TESTBEAT  — Material MIT stabilem Puls (Tempo-Map). Auf dem kurzen
#                      Schnipsel meldet die Erkennung zu Recht "kein stabiler Puls".
#
# Aufruf:
#   AIZANA_TESTKURZ=/path/to/fixtures/testmaterial_kurz.wav \
#   AIZANA_TESTBEAT=/tmp/kick121.wav \
#     ./scripts/uitest.sh lauf scripts/uitest-drehbuecher/27-clip-menue-block.txt

# ================================================== ET-L Merge AUDIO (Cmd+J)
# Erst in einen Clip klicken: ohne Tastaturfokus im Arrangement kommen Cmd+A und
# Cmd+J gar nicht an (der Befehl meldet dann "Taste nicht angenommen").
klickin Spurbereich Track 1 @ 0.10 0.5
warte 800
taste a cmd
warte 1000
taste j cmd
warte 8000
# Der Audio-Weg RENDERT, der zusammengefasste Clip heisst deshalb "… (Merged)".
> find Clip testmaterial_kurz (Merged)
> find Clip testmaterial_kurz
bild C-01-merged
menu 2 101
warte 2500
# Erwartung: beide Originale zurueck, der Merge-Clip weg
> find Clip testmaterial_kurz
> find Clip testmaterial_kurz (Merged)
bild C-02-merge-undo

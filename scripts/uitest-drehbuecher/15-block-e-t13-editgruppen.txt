# T1.3-L — Edit-Gruppen: Split propagiert auf die Spuren derselben Gruppe.
#
# Messgroesse: "find Clip testmaterial_kurz" meldet mehrdeutig MIT Anzahl UND mit
# den Bounds jedes Treffers — daran ist ablesbar, WELCHE Spur geteilt wurde (die
# y-Koordinate) und nicht nur, wie viele Clips es gibt.
#
# GEGENKONTROLLE eingebaut: derselbe Split OHNE Gruppenzuordnung darf nur EINEN
# Clip teilen. Ohne diesen Vergleich wuerde ein Split, der immer alles teilt,
# genauso bestehen.
#
# Kontextmenue-Eintraege sind ueber ihren TEXT ansprechbar (der Hook liest als
# letzten Ausweg den Titel des Accessibility-Handlers; PopupMenu-Items tragen ihn
# nur dort). Untermenues oeffnen ein EIGENES Fenster — danach "> peers" pruefen.
#
# Aufruf:
#   AIZANA_TESTKURZ=~/Desktop/"Aizana Testläufe"/2026-08-12/testmaterial_kurz.wav \
#     ./scripts/uitest.sh lauf scripts/uitest-drehbuecher/15-block-e-t13-editgruppen.txt

vorn
warte 2500
wennfehlt Spur hinzufuegen @ Pro
warte 1500

# Zweite Spur
klick Spur hinzufuegen
warte 1000
klick Audio
warte 400
klick Add
warte 2500

# Wo liegen die beiden Spuren? (Mit offenem Dock ist die zweite nur teilweise
# sichtbar — deshalb messen statt annehmen.)
> find Spurbereich Track 1
> find Spurbereich Audio

# Zeitgleiche Clips auf beiden Spuren
dropin Spurbereich Track 1 @ 0.20 0.5 $AIZANA_TESTKURZ
warte 2500
dropin Spurbereich Audio @ 0.20 0.5 $AIZANA_TESTKURZ
warte 2500
bild T13-00-zwei-clips
# Erwartung: mehrdeutig 2x, mit UNTERSCHIEDLICHEN y-Werten (je eine Spur)
> find Clip testmaterial_kurz

# ---------------------------------------------- GEGENKONTROLLE: OHNE Gruppe
# Geteilt wird mit dem SCHERENWERKZEUG (Taste C) — einen Menuepunkt "Split" gibt
# es nicht (MainComponent.h MenuIDs nachgeschlagen), und Cmd+E gehoert der
# nativen Menueleiste.
taste c
warte 400
klickin Spurbereich Track 1 @ 0.25 0.5
warte 1200
bild T13-01-split-ohne-gruppe
# Erwartung: 3x — zwei Teile auf Spur 1, Spur 2 unberuehrt. NICHT 4x.
> find Clip testmaterial_kurz

menu 2 101
warte 1200
# Erwartung: wieder 2x
> find Clip testmaterial_kurz

# ---------------------------------------------- beide Spuren in Edit-Gruppe 1
taste escape
warte 300
# PopupMenus lassen sich NICHT anklicken: MenuWindow prueft die ECHTE
# Systemmausposition (getCurrentRawMousePosition), die synthetische Ereignisse nie
# bewegen — Eintraege erster Ebene UND Untermenues bleiben tot. "press <Name>"
# loest den Eintrag stattdessen am Modell aus (AccessibilityActionType::press,
# prozessintern, kein Systemzugriff) und oeffnet Untermenues.
rklick Spur Track 1
warte 1000
> press Edit-Gruppe
warte 1000
> press Gruppe 1
warte 1000
bild T13-02-gruppe-spur1

rklick Spur Audio
warte 1000
> press Edit-Gruppe
warte 1000
> press Gruppe 1
warte 1000
bild T13-03-gruppe-spur2

# ---------------------------------------------- Split MIT Gruppe
taste c
warte 400
klickin Spurbereich Track 1 @ 0.25 0.5
warte 1500
bild T13-04-split-mit-gruppe
# Erwartung: 4x — der Schnitt sitzt auf BEIDEN Spuren
> find Clip testmaterial_kurz

# EIN Undo muss beide Schnitte zuruecknehmen
menu 2 101
warte 1500
bild T13-05-nach-einem-undo
# Erwartung: wieder 2x
> find Clip testmaterial_kurz

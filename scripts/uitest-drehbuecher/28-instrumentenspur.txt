#   ./scripts/uitest.sh lauf scripts/uitest-drehbuecher/28-instrumentenspur.txt

warte 2000

# --- Instrumentenspur ueber das Arrangement-Kontextmenue anlegen ------------
# (Der "Spur hinzufuegen"-Dialog ist ein DialogWindow; PopupMenus und Dialoge
#  brauchen `press`, synthetische Klicks erreichen sie nicht.)
rklickin Spurbereich Track 1 @ 0.5 0.9
warte 1200
> press Instrumentenspur hinzufügen (Synth)
warte 2500

# Die neue Spur muss da sein und einen MIDI-Clip tragen.
> bild 0 /tmp/aizana_gui_inst/inst-01-spur.png
> tree 0 14

# --- Scharfschalten geht (der Knopf existiert und nimmt den Klick an) -------
klick Aufnahmebereit Synth 2
warte 800

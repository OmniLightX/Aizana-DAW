# Voraussetzung: AIZANA_TESTWAV auf eine eigene WAV-Testdatei setzen.
# Block A — E-V1/E-V2/E-V4/E-V10 + Nachfrage zur fehlenden Wellenform im Arrangement.
# Aufruf: ./scripts/uitest.sh lauf scripts/uitest-drehbuecher/03-editor-werkzeuge.txt

# Ansicht HERSTELLEN, nicht voraussetzen: der Modus ist persistiert, und ein hart
# beendeter Lauf schreibt ihn nicht zurueck. Blind auf 'Pro' zu klicken wuerde in
# der Pro-Ansicht nichts finden und den Lauf abbrechen.
wennfehlt Spurbereich Track 1 @ Pro
warte 2500
dropin Spurbereich Track 1 @ 0.40 0.5 $AIZANA_TESTWAV
warte 3000

# Wellenform im ARRANGEMENT: erscheint sie spaeter? (im Dock ist sie sofort da)
bild 30-arrangement-direkt
warte 8000
bild 31-arrangement-nach-8s

klick Clip 121 bpm_809
warte 600
# Dock-Zustand ist persistiert — nur oeffnen, wenn die Werkzeugleiste fehlt.
wennfehlt Werkzeug Pfeil @ Unten Editor
warte 2000
bild 32-dock-offen

# --- E-V2 Split: Werkzeug Teilen waehlen, in der Mitte des Clips klicken ---
klick Werkzeug Teilen
warte 500
bild 33-werkzeug-teilen-aktiv
klickin Clip 121 bpm_809 @ 0.5 0.5
warte 1200
bild 34-nach-split
namen

# --- E-V4 Mute: Clip stummschalten ---
klick Werkzeug Pfeil
warte 400
klick Clip stummschalten
warte 1000
bild 35-clip-stumm

# Rueckgaengig, damit der Zustand fuer die naechsten Punkte sauber ist
# Cmd-Kuerzel gehoeren auf macOS der NATIVEN Menueleiste — ein synthetischer
# Tastendruck an JUCE erreicht sie nicht. Deshalb ueber das Menue-Modell:
# Menue 2 = Edit, Punkt 101 = Undo.
menu 2 101
warte 800
menu 2 101
warte 800
bild 36-nach-undo
